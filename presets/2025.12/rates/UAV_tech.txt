#$ DESCRIPTION:
#$ DESCRIPTION: Information:
#$ DESCRIPTION: -----------
#$ DESCRIPTION: FREESTYLE: Actual rates with a maximum of 1000 degrees/sec for roll and pitch, and 700 degrees/sec for yaw. Smooth around center stick, but still snappy at the edges.
#$ DESCRIPTION:
#$ DESCRIPTION: Applies to the currently selected rate profile. The Whoop option also changes Angle/Horizon settings in the currently selected PID profile.
#$ DESCRIPTION:
#$ DESCRIPTION: OPTIONS:
#$ DESCRIPTION: -----------
#$ DESCRIPTION: - CINEMATIC/HD: Keeps the same maximum rates and adds more expo around center stick for smoother transitions and improved HD footage.
#$ DESCRIPTION: - WHOOP: For whoop racing in Angle/Horizon modes. Sets yaw to 600 degrees/sec with 25% expo, Angle P gain and Horizon leveling strength to 70, the Angle limit to 80 degrees, and the Horizon stick limit to 100%. Also sets 50% throttle expo with the throttle midpoint and hover point at 45%.
#$ DESCRIPTION:
#$ DESCRIPTION: If both options are selected, the Whoop yaw settings override the Cinematic/HD yaw settings; roll and pitch retain the Cinematic/HD expo.
#$ DESCRIPTION:
#$ DESCRIPTION: Angle/Horizon controllers and throttle curves have changed since Betaflight 4.5, so the Whoop option does not reproduce the older preset's behavior exactly.
#$ DESCRIPTION:
#$ DISCUSSION: https://github.com/betaflight/firmware-presets/pull/609
#$ INCLUDE: presets/2025.12/rates/defaults.txt

set rates_type = ACTUAL
set roll_rc_rate = 1
set pitch_rc_rate = 1
set yaw_rc_rate = 1
set roll_expo = 55
set pitch_expo = 55
set yaw_expo = 55
set roll_srate = 100
set pitch_srate = 100
set yaw_srate = 70

#$ OPTION BEGIN (UNCHECKED): Cinematic/HD Smoothing
    # -- Cinematic/HD expo --
    set roll_expo = 70
    set pitch_expo = 70
    set yaw_expo = 70
#$ OPTION END

#$ OPTION BEGIN (UNCHECKED): Whooping in Angle/Horizon Modes
    # -- Whoop yaw, throttle, and Angle/Horizon settings --
    set thr_mid = 45
    set thr_expo = 50
    set thr_hover = 45
    set yaw_rc_rate = 10
    set yaw_expo = 25
    set yaw_srate = 60
    set angle_p_gain = 70
    set horizon_level_strength = 70
    set horizon_limit_sticks = 100
    set angle_limit = 80
#$ OPTION END
